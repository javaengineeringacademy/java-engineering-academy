# Template Method Pattern in Python

The Template Method pattern defines the skeleton of an algorithm in a base class, allowing subclasses to override specific steps without changing the algorithm's structure. Python's abstract methods and hooks make this pattern clean.

## When to Use

- Multiple classes share similar algorithm structure
- Common behavior should be factored into a single location
- Subclasses should customize only parts of an algorithm
- Avoiding code duplication in similar classes
- Framework design and extensibility

## Python Implementation

### Abstract Base Class
```python
from abc import ABC, abstractmethod

class DataMiner(ABC):
    def mine(self, path: str):
        data = self.extract(path)
        parsed = self.parse(data)
        filtered = self.filter(parsed)
        analysis = self.analyze(filtered)
        report = self.report(analysis)
        return report

    @abstractmethod
    def extract(self, path: str) -> str:
        pass

    @abstractmethod
    def parse(self, data: str) -> list:
        pass

    def filter(self, data: list) -> list:
        return [d for d in data if d]

    def analyze(self, data: list) -> dict:
        return {"count": len(data), "data": data}

    def report(self, analysis: dict) -> str:
        return f"Report: {analysis['count']} items"

class CSVMiner(DataMiner):
    def extract(self, path: str) -> str:
        return f"CSV data from {path}"

    def parse(self, data: str) -> list:
        return data.split(",")

class JSONMiner(DataMiner):
    def extract(self, path: str) -> str:
        return f"JSON data from {path}"

    def parse(self, data: str) -> list:
        return [data]
```

### Hook Methods
```python
class GameAI(ABC):
    def turn(self):
        self.collect_resources()
        self.build_structures()
        self.train_units()
        if self.should_attack():
            self.attack()

    def collect_resources(self):
        print("Collecting resources")

    def build_structures(self):
        print("Building structures")

    def train_units(self):
        print("Training units")

    @abstractmethod
    def attack(self):
        pass

    def should_attack(self) -> bool:
        return True  # Hook with default

class AggressiveAI(GameAI):
    def attack(self):
        print("Attacking aggressively")

    def should_attack(self):
        return True

class DefensiveAI(GameAI):
    def attack(self):
        print("Defending territory")

    def should_attack(self):
        return False
```

### Using Decorators
```python
def template_method(*steps):
    def decorator(cls):
        original_init = cls.__init__

        def new_init(self, *args, **kwargs):
            original_init(self, *args, **kwargs)
            self._steps = steps

        cls.__init__ = new_init

        def execute(self):
            results = []
            for step in self._steps:
                results.append(getattr(self, step)())
            return results

        cls.execute = execute
        return cls
    return decorator

@template_method("step_one", "step_two", "step_three")
class MyProcess:
    def step_one(self):
        return "Step 1 done"

    def step_two(self):
        return "Step 2 done"

    def step_three(self):
        return "Step 3 done"
```

## Pythonic Alternative

Use function composition for simple templates:
```python
def template(*functions):
    def wrapper(data):
        result = data
        for func in functions:
            result = func(result)
        return result
    return wrapper

pipeline = template(
    lambda x: x.strip(),
    lambda x: x.lower(),
    lambda x: x.replace(" ", "_")
)

print(pipeline("  Hello World  "))  # hello_world
```

## Real-World Example

```python
class ReportGenerator(ABC):
    def generate(self, data: dict):
        header = self.create_header(data)
        body = self.create_body(data)
        footer = self.create_footer(data)
        return f"{header}\n{body}\n{footer}"

    @abstractmethod
    def create_header(self, data: dict) -> str:
        pass

    @abstractmethod
    def create_body(self, data: dict) -> str:
        pass

    def create_footer(self, data: dict) -> str:
        return "Generated by Report System"

class SalesReport(ReportGenerator):
    def create_header(self, data):
        return f"Sales Report - {data.get('period', 'N/A')}"

    def create_body(self, data):
        return f"Total Sales: ${data.get('total', 0):,.2f}"
```

## Best Practices

1. Keep template methods focused on orchestration
2. Use hook methods for optional customization points
3. Document which methods are meant to be overridden
4. Consider using Protocol for structural typing
5. Avoid deep template hierarchies
